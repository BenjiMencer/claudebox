# ~/.zshrc

# ---------------------------------------------------------------------------
# Basic shell niceties (optional — trim to taste)
# ---------------------------------------------------------------------------
export EDITOR="vim"
export CLICOLOR=1
setopt AUTO_CD                      # type a dir name to cd into it
setopt HIST_IGNORE_DUPS
HISTFILE="$HOME/.zsh_history"
HISTSIZE=10000
SAVEHIST=10000

# ---------------------------------------------------------------------------
# claude-local: contained, dockerized Claude Code
#
# Where you cloned the repo. This is the only path to edit — everything below
# is derived from it.
# ---------------------------------------------------------------------------
CLAUDEBOX_REPO="$HOME/claude-docker"

# Source the function from the repo rather than defining it here, so a
# `git pull` actually updates it. Don't paste the function body into .zshrc:
# a pasted copy silently ignores every future pull, and you end up rebuilding
# the image from new code while the launcher around it stays old.
[ -f "$CLAUDEBOX_REPO/claude-local.zsh" ] && source "$CLAUDEBOX_REPO/claude-local.zsh"
